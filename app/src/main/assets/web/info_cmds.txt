#### pre
r(){ printf '%s|%s\n' "$1" "$2"; }
h(){ printf '## %s\n' "$1"; }
bar(){ printf '!B|%s|%s|%s\n' "$1" "$2" "$3"; }
rd(){ [ -r "$1" ] && head -n 1 "$1" 2>/dev/null | tr -d '\r'; }
rr(){ _v=$(rd "$2"); [ -n "$_v" ] && r "$1" "$_v$3"; }
gp(){ getprop "$1" 2>/dev/null; }
rp(){ _v=$(gp "$2"); [ -n "$_v" ] && r "$1" "$_v"; }
pp(){ getprop 2>/dev/null | grep -iE "$1" | sed 's/^\[\(.*\)\]: \[\(.*\)\]$/\1|\2/' | head -n "${2:-40}"; }
num(){ case "$1" in ''|*[!0-9]*) return 1;; esac; return 0; }
kb(){ num "$1" || { echo "$1"; return; }; if [ "$1" -ge 1048576 ]; then echo "$(($1/1048576)).$((($1%1048576)*10/1048576)) GB"; elif [ "$1" -ge 1024 ]; then echo "$(($1/1024)) MB"; else echo "$1 kB"; fi; }
hb(){ num "$1" || { echo "$1"; return; }; if [ "$1" -ge 1073741824 ]; then echo "$(($1/1073741824)).$((($1%1073741824)*100/1073741824)) GB"; elif [ "$1" -ge 1048576 ]; then echo "$(($1/1048576)) MB"; elif [ "$1" -ge 1024 ]; then echo "$(($1/1024)) KB"; else echo "$1 B"; fi; }
mhz(){ num "$1" || { echo "-"; return; }; echo "$(($1/1000)) MHz"; }
tc(){ num "$1" || { echo "$1"; return; }; if [ "$1" -ge 1000 ]; then echo "$(($1/1000)).$((($1%1000)/100)) °C"; else echo "$1 °C"; fi; }
dc(){ _o=$(printf '%s\n' "$DM" | grep -oiE "$2" | tr 'A-Z' 'a-z' | sort -u | head -n 12 | tr '\n' ' '); [ -n "$_o" ] && r "$1" "$_o"; }
